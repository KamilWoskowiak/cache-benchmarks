213 cache_manager(size_type capacity, mapping_type mapping, policy_type pol, summarizer_type summarizer = {})
214 : capacity_{capacity}, pending_index_{capacity + 1}, mapping_{mapping}, summarizer_{summarizer}, policy_{pol},
215 entry_values_(capacity), entry_ref_counts_(capacity), entry_metadata_(capacity),
216 entry_occupied_(capacity, false), heap_(heap_compare{&entry_metadata_}), entry_heap_handles_(capacity),
217 entry_heap_prio_stale(capacity, false) {
218 dirty_slots_stack_.reserve(capacity_);
219 deferred_slots_.reserve(capacity_);

222 entry_metadata_[slot].key = key_type{};

239 auto hit_percent() const noexcept -> double {
240 const std::size_t lookups = lookup_count_.load(std::memory_order_relaxed);
241
242 if (lookups == 0) {
243 return 0.0;
244 }
245
246 const std::size_t hits = hit_count_.load(std::memory_order_relaxed);
247
249 return (100.0 * gsl::narrow_cast<double>(hits)) / gsl::narrow_cast<double>(lookups);

260 [[nodiscard]]
261 auto lookup(key_type key) const -> std::optional<handle_type> {
262 // lookup_count_.fetch_add(1, std::memory_order_relaxed);
263 std::size_t slot{pending_index_};
264
265 // Captures the mapped slot and pins it before returning a handle.
266 auto record_slot = [this, &slot](const index_map_type::value_type& kv) {
267 slot = kv.second;
268 if (slot != pending_index_) {
269 // hit_count_.fetch_add(1, std::memory_order_relaxed);
270 entry_ref_counts_[slot].value.fetch_add(1, std::memory_order_relaxed);
271 }
272 };
273
274 // check if there is a valid mapping for key
275 // this key is still being loaded, wait. after it is done loading we call try_emplace_l, if the key is present
276 // we return a handle. if it is not present, we take responsibility for constructing it
277 bool inserted{false};
278 while (!inserted) {
279
280 if (key_to_index_.if_contains(key, record_slot)) {
281 if (slot == pending_index_) {
282 m_wait_load_(key);
283 } else {
284 return m_make_handle_(slot);
285 }
286 }
287
288 if (!key_to_index_.try_emplace_l(key, record_slot, pending_index_)) {
289 if (slot == pending_index_) {
290 m_wait_load_(key);
291 } else {
292 return m_make_handle_(slot);
293 }
294 } else {
295 inserted = true;
296 }
297 }
298
299 // if we are at this point (inserted = true), we are inserting a new cache entry
300 m_init_latch_(key);
301
302 waiters_.fetch_add(1, std::memory_order_acq_rel);
303
304 m_produce_slots_();
305
306 value_type loaded_value{};
307
308 auto error = mapping_(key, loaded_value);
309
310 if (error.has_value()) {
311 m_fail_load_(key);
312 // TODO: we need a way to return std::error to the user
313 return std::nullopt;
314 }
315
316 std::size_t free_slot{pending_index_};
317 free_slots_.pop(free_slot);
318
319 if (free_slot == pending_index_) {
320 m_fail_load_(key);
321 return std::nullopt;
322 }
323
324 return m_fill_slot_(key, free_slot, std::move(loaded_value));
325 }

468 auto m_init_latch_(key_type key) const -> void {
469 key_to_latch_.try_emplace_l(
470 key, [](auto&) {}, // already exists, no-op
471 std::make_shared<key_latch>());
472 }
473
475 auto m_wait_load_(key_type key) const -> bool {
476 std::shared_ptr<key_latch> latch;
477
478 bool contained = key_to_latch_.if_contains(key, [&](auto& it) { latch = it.second; });
479
480 // The latch disappeared before we waited; the load either failed or has already published a valid slot.
481 if (!contained) {
482 return false;
483 }
484
485 latch->is_ready.wait(false, std::memory_order_acquire);
486 return true;
487 }
488
490 auto m_finish_load_(key_type key, std::size_t slot, bool slot_is_valid) const -> void {
491 if (slot_is_valid) {
492 key_to_index_.modify_if(key, [&](index_map_type::value_type& kv) { kv.second = slot; });
493 } else {
494 key_to_index_.erase_if(key, [&](const index_map_type::value_type& kv) { return kv.second == pending_index_; });
495 }
496
497 std::shared_ptr<key_latch> latch;
498
499 key_to_latch_.erase_if(key, [&](auto& it) {
500 latch = std::move(it.second);
501 return true;
502 });
503
504 if (!latch) {
505 return;
506 }
507
508 latch->is_ready.store(true, std::memory_order_release);
509 latch->is_ready.notify_all();
510 }
511
513 auto m_fail_load_(const key_type& key) const -> void { m_finish_load_(key, pending_index_, false); }

516 auto m_produce_slots_() const -> void {
517 oneapi::tbb::spin_mutex::scoped_lock lock(evict_mutex_);
518
519 const std::size_t pending = waiters_.load(std::memory_order_acquire);
520
521 if (pending == 0) {
522 return;
523 }
524
525 const std::size_t batch = std::min(pending, eviction_batch_size);
526
527 waiters_.fetch_sub(batch, std::memory_order_acq_rel);
528
529 m_drain_heap_();
530
531 for (std::size_t i = 0; i < batch; ++i) {
532 free_slots_.push(m_evict_slot_());
533 }
534 }

546 [[nodiscard]]
547 auto m_fill_slot_(const key_type& key, std::size_t slot, value_type value) const -> std::optional<handle_type> {
548 entry_ref_counts_[slot].value.fetch_add(1, std::memory_order_relaxed);
549
550 metadata& entry = entry_metadata_[slot];
551
552 entry.key = key;
553 entry.priority = policy_.identity;
554
555 entry_values_[slot] = std::move(value);
556
557 {
558 oneapi::tbb::spin_mutex::scoped_lock lock(evict_mutex_);
559
560 entry_occupied_[slot] = true;
561 deferred_slots_.push_back(slot);
562 }
563
564 m_finish_load_(key, slot, true);
565
566 return m_make_handle_(slot);
567 }

570 auto m_merge_accesses_(typename log_type::iterator begin, typename log_type::iterator end) const -> void {
571 std::vector<std::size_t> slots;
572
573 for (auto it = begin; it != end; ++it) {
574 slots.push_back(*it);
575 }
576
577 if (slots.empty()) {
578 return;
579 }
580
581 oneapi::tbb::spin_mutex::scoped_lock lock(log_mutex_);
582
583 for (const std::size_t slot : slots) {
584 metadata& entry = entry_metadata_[slot];
585
586 summary_type summary = summarizer_(entry.key);
587 priority_type access_prio = policy_.prioritizer(clock_++, summary);
588
589 entry.priority = policy_.combiner(entry.priority, access_prio);
590
591 if (!entry_heap_prio_stale[slot]) {
592 entry_heap_prio_stale[slot] = true;
593 dirty_slots_stack_.push_back(slot);
594 }
595 }
596 }
597
599 auto m_drain_heap_() const -> void {
600 access_log_.drain([this](log_type::iterator begin, log_type::iterator end) { m_merge_accesses_(begin, end); });
601
602 oneapi::tbb::spin_mutex::scoped_lock lock(log_mutex_);
603
604 for (const std::size_t slot : deferred_slots_) {
605 if (!entry_occupied_[slot]) {
606 continue;
607 }
608
609 heap_link& meta = entry_heap_handles_[slot];
610 meta.handle = heap_.push(slot);
611 }
612
613 deferred_slots_.clear();
614
615 for (const std::size_t slot : dirty_slots_stack_) {
616 entry_heap_prio_stale[slot] = false;
617
618 if (!entry_occupied_[slot]) {
619 continue;
620 }
621
622 heap_link& meta = entry_heap_handles_[slot];
623 heap_.update(meta.handle);
624 }
625
626 dirty_slots_stack_.clear();
627 }
628
630 auto m_evict_slot_() const -> std::size_t {
631 while (!heap_.empty()) {
632 const std::size_t slot = heap_.top();
633 heap_.pop();
634
635 metadata& slot_entry = entry_metadata_[slot];
636
637 // Empty slot. It is already removed from the heap and will be reinserted after the miss owner fills it.
638 if (!entry_occupied_[slot]) {
639 return slot;
640 }
641
642 auto can_evict = [&](const index_map_type::value_type& kv) {
643 return kv.second == slot && entry_ref_counts_[slot].value.load(std::memory_order_acquire) == 0;
644 };
645
646 if (key_to_index_.erase_if(slot_entry.key, can_evict)) {
647 entry_occupied_[slot] = false;
648
649 slot_entry.key = key_type{};
650 slot_entry.priority = policy_.identity;
651
652 return slot;
653 }
654
655 deferred_slots_.push_back(slot);
656 }
657
658 return pending_index_;
659 }
